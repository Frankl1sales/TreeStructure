class Node {
  data: string;  // Aqui vai o nome do animal ou grupo de animais
  left: Node | null;
  right: Node | null;

  constructor(data: string) {
      this.data = data;
      this.left = null;
      this.right = null;
  }
}
  
function createTree(): Node {
  const root = new Node("Animalia");
  root.left = new Node("Chordata");
  root.left.left = new Node("Mammalia");
  root.left.left.left = new Node("Primates");
  root.left.left.left.left = new Node("Homo sapiens");
  root.left.left.right = new Node("Carnivora");
  root.left.left.right.left = new Node("Canis lupus");
  root.left.right = new Node("Aves");
  root.left.right.left = new Node("Passeriformes");
  root.left.right.left.left = new Node("Corvus corax");
  return root; // Retorna a árvore pronta 
}

// Criando a árvore 
const rootNode = createTree();
